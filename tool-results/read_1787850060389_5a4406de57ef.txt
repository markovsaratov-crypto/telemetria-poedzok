     2→
     3→// src/components/sessions-list.tsx — список сессий с курсорной пагинацией, фильтрами, сортировкой, view modes.
     4→
     5→import * as React from "react";
     6→import { motion, AnimatePresence } from "framer-motion";
     7→import {
     8→  Search,
     9→  Loader2,
    10→  ChevronRight,
    11→  Filter,
    12→  X,
    13→  MapPin,
    14→  Clock,
    15→  Activity,
    16→  ArrowDownUp,
    17→  LayoutList,
    18→  Rows3,
    19→  HardDrive,
    20→  Calendar,
    21→  RefreshCw,
    22→  Trash2,
    23→  CheckSquare,
    24→  Square,
    25→} from "lucide-react";
    26→import { toast } from "sonner";
    27→import { useSessions, useRoutes, useBulkDeleteSessions, useDeleteSession } from "@/lib/hooks";
    28→import { Button } from "@/components/ui/button";
    29→import { Input } from "@/components/ui/input";
    30→import {
    31→  Select,
    32→  SelectContent,
    33→  SelectItem,
    34→  SelectTrigger,
    35→  SelectValue,
    36→} from "@/components/ui/select";
    37→import { Badge } from "@/components/ui/badge";
    38→import { Skeleton } from "@/components/ui/skeleton";
    39→import { fmtDate, fmtBytes, fmtNumber } from "@/lib/format";
    40→import type { SessionListItem } from "@/lib/api-client";
    41→import { cn } from "@/lib/utils";
    42→
    43→interface SessionsListProps {
    44→  selectedId: string | null;
    45→  onSelect: (id: string) => void;
    46→}
    47→
    48→type SortKey = "date_desc" | "date_asc" | "points_desc" | "size_desc";
    49→type ViewMode = "detailed" | "compact";
    50→
    51→const STATUS_LABEL: Record<string, string> = {
    52→  active: "Активна",
    53→  completed: "Завершена",
    54→  archived: "Архив",
    55→  deleted: "Удалена",
    56→};
    57→
    58→const STATUS_BADGE: Record<string, string> = {
    59→  active: "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/30",
    60→  completed: "bg-teal-500/15 text-teal-700 dark:text-teal-400 border-teal-500/30",
    61→  archived: "bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/30",
    62→  deleted: "bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/30",
    63→};
    64→
    65→const SORT_LABELS: Record<SortKey, string> = {
    66→  date_desc: "Сначала новые",
    67→  date_asc: "Сначала старые",
    68→  points_desc: "Больше точек",
    69→  size_desc: "Больше объём",
    70→};
    71→
    72→export function SessionsList({ selectedId, onSelect }: SessionsListProps) {
    73→  const [deviceId, setDeviceId] = React.useState("");
    74→  const [status, setStatus] = React.useState<string>("");
    75→  const [routeId, setRouteId] = React.useState<string>("");
    76→  const [cursor, setCursor] = React.useState<string | undefined>(undefined);
    77→  const [allSessions, setAllSessions] = React.useState<SessionListItem[]>([]);
    78→  const [hasMore, setHasMore] = React.useState(false);
    79→  const [sort, setSort] = React.useState<SortKey>("date_desc");
    80→  const [viewMode, setViewMode] = React.useState<ViewMode>("detailed");
    81→  const [bulkMode, setBulkMode] = React.useState(false);
    82→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
    83→
    84→  const bulkDeleteMut = useBulkDeleteSessions();
    85→  const deleteMut = useDeleteSession();
    86→  const [deletingId, setDeletingId] = React.useState<string | null>(null);
    87→
    88→  // Применённые фильтры
    89→  const [applied, setApplied] = React.useState<{
    90→    deviceId?: string;
    91→    status?: string;
    92→    routeId?: string;
    93→  }>({});
    94→
    95→  const { data, isLoading, isFetching, refetch } = useSessions({
    96→    limit: 20,
    97→    cursor,
    98→    deviceId: applied.deviceId,
    99→    status: applied.status,
   100→    routeId: applied.routeId,
   101→  });
   102→
   103→  const { data: routesData } = useRoutes();
   104→
   105→  React.useEffect(() => {
   106→    setAllSessions([]);
   107→    setCursor(undefined);
   108→  }, [applied]);
   109→
   110→  React.useEffect(() => {
   111→    if (data) {
   112→      if (cursor) {
   113→        setAllSessions((prev) => [...prev, ...data.sessions]);
   114→      } else {
   115→        setAllSessions(data.sessions);
   116→      }
   117→      setHasMore(!!data.nextCursor);
   118→    }
   119→  }, [data, cursor]);
   120→
   121→  // Сортировка на клиенте (после загрузки)
   122→  const sortedSessions = React.useMemo(() => {
   123→    const arr = [...allSessions];
   124→    arr.sort((a, b) => {
   125→      switch (sort) {
   126→        case "date_asc":
   127→          return new Date(a.startTime).getTime() - new Date(b.startTime).getTime();
   128→        case "points_desc":
   129→          return b.pointCount - a.pointCount;
   130→        case "size_desc":
   131→          return b.payloadBytes - a.payloadBytes;
   132→        default:
   133→          return new Date(b.startTime).getTime() - new Date(a.startTime).getTime();
   134→      }
   135→    });
   136→    return arr;
   137→  }, [allSessions, sort]);
   138→
   139→  // Группировка по дате (только для detailed view)
   140→  const grouped = React.useMemo(() => {
   141→    if (viewMode !== "detailed") return null;
   142→    const groups: { label: string; items: SessionListItem[] }[] = [];
   143→    const today = new Date();
   144→    today.setHours(0, 0, 0, 0);
   145→    const yesterday = new Date(today);
   146→    yesterday.setDate(yesterday.getDate() - 1);
   147→
   148→    for (const s of sortedSessions) {
   149→      const d = new Date(s.startTime);
   150→      d.setHours(0, 0, 0, 0);
   151→      let label: string;
   152→      if (d.getTime() === today.getTime()) label = "Сегодня";
   153→      else if (d.getTime() === yesterday.getTime()) label = "Вчера";
   154→      else
   155→        label = d.toLocaleDateString("ru-RU", {
   156→          day: "numeric",
   157→          month: "long",
   158→          year: d.getFullYear() === today.getFullYear() ? undefined : "numeric",
   159→        });
   160→
   161→      let g = groups.find((g) => g.label === label);
   162→      if (!g) {
   163→        g = { label, items: [] };
   164→        groups.push(g);
   165→      }
   166→      g.items.push(s);
   167→    }
   168→    return groups;
   169→  }, [sortedSessions, viewMode]);
   170→
   171→  function applyFilters() {
   172→    setApplied({
   173→      deviceId: deviceId.trim() || undefined,
   174→      status: status || undefined,
   175→      routeId: routeId || undefined,
   176→    });
   177→  }
   178→
   179→  function resetFilters() {
   180→    setDeviceId("");
   181→    setStatus("");
   182→    setRouteId("");
   183→    setApplied({});
   184→  }
   185→
   186→  function toggleSelect(id: string) {
   187→    setSelectedIds((prev) => {
   188→      const next = new Set(prev);
   189→      if (next.has(id)) next.delete(id);
   190→      else next.add(id);
   191→      return next;
   192→    });
   193→  }
   194→
   195→  function selectAll() {
   196→    setSelectedIds(new Set(allSessions.map((s) => s.id)));
   197→  }
   198→
   199→  function clearSelection() {
   200→    setSelectedIds(new Set());
   201→  }
   202→
   203→  async function handleBulkDelete() {
   204→    if (selectedIds.size === 0) return;
   205→    if (!confirm(`Удалить ${selectedIds.size} сессий? Это soft-delete с grace period 30 дней.`)) {
   206→      return;
   207→    }
   208→    try {
   209→      const result = await bulkDeleteMut.mutateAsync(Array.from(selectedIds));
   210→      toast.success(`Удалено: ${result.deleted}`, {
   211→        description: result.errors.length > 0 ? `${result.errors.length} не найдено` : undefined,
   212→      });
   213→      clearSelection();
   214→      setBulkMode(false);
   215→    } catch (e) {
   216→      toast.error("Ошибка массового удаления", { description: (e as Error).message });
   217→    }
   218→  }
   219→
   220→  async function handleQuickDelete(e: React.MouseEvent, s: SessionListItem) {
   221→    e.stopPropagation();
   222→    e.preventDefault();
   223→    const label = s.deviceName || s.deviceId;
   224→    const date = new Date(s.startTime).toLocaleDateString("ru-RU", { day: "numeric", month: "short", year: "numeric" });
   225→    if (!confirm(`Удалить поездку?\n\n${label} · ${date} · ${fmtNumber(s.pointCount)} точек\n\nЭто soft-delete с grace period 30 дней.`)) {
   226→      return;
   227→    }
   228→    setDeletingId(s.id);
   229→    try {
   230→      await deleteMut.mutateAsync(s.id);
   231→      toast.success("Поездка удалена", {
   232→        description: `${label} · grace period 30 дней, потом purge`,
   233→      });
   234→      // Если удалили текущую выбранную — сбрасываем
   235→      if (selectedId === s.id) {
   236→        onSelect("");
   237→      }
   238→    } catch (err) {
   239→      toast.error("Ошибка удаления", { description: (err as Error).message });
   240→    } finally {
   241→      setDeletingId(null);
   242→    }
   243→  }
   244→
   245→  const activeFiltersCount =
   246→    (applied.deviceId ? 1 : 0) + (applied.status ? 1 : 0) + (applied.routeId ? 1 : 0);
   247→
   248→  return (
   249→    <div className="flex flex-col h-full">
   250→      {/* Header с count и view toggle */}
   251→      <div className="border-b px-3 py-2 flex items-center justify-between bg-muted/30 gap-2">
   252→        <div className="flex items-center gap-2 text-xs">
   253→          <span className="font-medium">
   254→            {allSessions.length > 0 ? `${allSessions.length} сессий` : "Сессии"}
   255→          </span>
   256→          {isFetching && (
   257→            <Loader2 className="h-3 w-3 animate-spin text-muted-foreground" />
   258→          )}
   259→        </div>
   260→        <div className="flex items-center gap-1">
   261→          <Button
   262→            variant="ghost"
   263→            size="sm"
   264→            className="h-7 w-7 p-0"
   265→            onClick={() => refetch()}
   266→            title="Обновить"
   267→          >
   268→            <RefreshCw className="h-3 w-3" />
   269→          </Button>
   270→          <div className="flex items-center rounded-md border bg-background/50 p-0.5">
   271→            <button
   272→              onClick={() => setViewMode("detailed")}
   273→              className={cn(
   274→                "p-1 rounded transition-colors",
   275→                viewMode === "detailed" ? "bg-primary/15 text-primary" : "text-muted-foreground hover:text-foreground"
   276→              )}
   277→              title="Детальный вид"
   278→            >
   279→              <Rows3 className="h-3 w-3" />
   280→            </button>
   281→            <button
   282→              onClick={() => setViewMode("compact")}
   283→              className={cn(
   284→                "p-1 rounded transition-colors",
   285→                viewMode === "compact" ? "bg-primary/15 text-primary" : "text-muted-foreground hover:text-foreground"
   286→              )}
   287→              title="Компактный вид"
   288→            >
   289→              <LayoutList className="h-3 w-3" />
   290→            </button>
   291→          </div>
   292→          <Button
   293→            variant="ghost"
   294→            size="sm"
   295→            className="h-7 gap-1 text-xs"
   296→            onClick={() => {
   297→              setBulkMode((v) => !v);
   298→              if (bulkMode) clearSelection();
   299→            }}
   300→            title={bulkMode ? "Выйти из режима выбора" : "Массовое удаление"}
   301→          >
   302→            {bulkMode ? (
   303→              <>
   304→                <X className="h-3 w-3" /> Отмена
   305→              </>
   306→            ) : (
   307→              <>
   308→                <CheckSquare className="h-3 w-3" /> Выбрать
   309→              </>
   310→            )}
   311→          </Button>
   312→        </div>
   313→      </div>
   314→
   315→      {/* Bulk actions bar */}
   316→      {bulkMode && (
   317→        <motion.div
   318→          initial={{ opacity: 0, height: 0 }}
   319→          animate={{ opacity: 1, height: "auto" }}
   320→          exit={{ opacity: 0, height: 0 }}
   321→          className="border-b bg-amber-500/5 px-3 py-2 flex items-center justify-between gap-2"
   322→        >
   323→          <div className="flex items-center gap-2 text-xs">
   324→            <span className="font-medium text-amber-700 dark:text-amber-400">
   325→              Выбрано: {selectedIds.size}
   326→            </span>
   327→            <button onClick={selectAll} className="text-[10px] underline text-muted-foreground hover:text-foreground">
   328→              все
   329→            </button>
   330→            <button onClick={clearSelection} className="text-[10px] underline text-muted-foreground hover:text-foreground">
   331→              очистить
   332→            </button>
   333→          </div>
   334→          <Button
   335→            size="sm"
   336→            variant="destructive"
   337→            className="h-7 gap-1 text-xs"
   338→            disabled={selectedIds.size === 0 || bulkDeleteMut.isPending}
   339→            onClick={handleBulkDelete}
   340→          >
   341→            {bulkDeleteMut.isPending ? (
   342→              <Loader2 className="h-3 w-3 animate-spin" />
   343→            ) : (
   344→              <Trash2 className="h-3 w-3" />
   345→            )}
   346→            Удалить ({selectedIds.size})
   347→          </Button>
   348→        </motion.div>
   349→      )}
   350→
   351→      {/* Фильтры */}
   352→      <div className="space-y-2 border-b p-3 bg-muted/20">
   353→        <div className="flex items-center justify-between">
   354→          <div className="flex items-center gap-2 text-xs font-medium text-muted-foreground">
   355→            <Filter className="h-3.5 w-3.5" /> Фильтры
   356→            {activeFiltersCount > 0 && (
   357→              <Badge className="text-[9px] h-4 px-1 bg-primary text-primary-foreground">
   358→                {activeFiltersCount}
   359→              </Badge>
   360→            )}
   361→          </div>
   362→          {/* Sort */}
   363→          <Select value={sort} onValueChange={(v) => setSort(v as SortKey)}>
   364→            <SelectTrigger className="h-7 w-[140px] text-[10px] gap-1">
   365→              <ArrowDownUp className="h-2.5 w-2.5" />
   366→              <SelectValue />
   367→            </SelectTrigger>
   368→            <SelectContent>
   369→              {(Object.keys(SORT_LABELS) as SortKey[]).map((k) => (
   370→                <SelectItem key={k} value={k} className="text-xs">
   371→                  {SORT_LABELS[k]}
   372→                </SelectItem>
   373→              ))}
   374→            </SelectContent>
   375→          </Select>
   376→        </div>
   377→        <div className="grid grid-cols-1 sm:grid-cols-12 gap-2">
   378→          <div className="sm:col-span-5 relative">
   379→            <Search className="absolute left-2 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   380→            <Input
   381→              placeholder="Device ID…"
   382→              value={deviceId}
   383→              onChange={(e) => setDeviceId(e.target.value)}
   384→              className="pl-7 h-8 text-xs"
   385→              onKeyDown={(e) => {
   386→                if (e.key === "Enter") applyFilters();
   387→              }}
   388→            />
   389→          </div>
   390→          <Select value={status || "all"} onValueChange={(v) => setStatus(v === "all" ? "" : v)}>
   391→            <SelectTrigger className="sm:col-span-3 h-8 text-xs">
   392→              <SelectValue placeholder="Статус" />
   393→            </SelectTrigger>
   394→            <SelectContent>
   395→              <SelectItem value="all">Все</SelectItem>
   396→              <SelectItem value="active">Активна</SelectItem>
   397→              <SelectItem value="completed">Завершена</SelectItem>
   398→              <SelectItem value="archived">Архив</SelectItem>
   399→            </SelectContent>
   400→          </Select>
   401→          <Select value={routeId || "all"} onValueChange={(v) => setRouteId(v === "all" ? "" : v)}>
   402→            <SelectTrigger className="sm:col-span-3 h-8 text-xs">
   403→              <SelectValue placeholder="Маршрут" />
   404→            </SelectTrigger>
   405→            <SelectContent>
   406→              <SelectItem value="all">Все маршруты</SelectItem>
   407→              {routesData?.routes.map((r) => (
   408→                <SelectItem key={r.id} value={r.id}>
   409→                  {r.name}
   410→                </SelectItem>
   411→              ))}
   412→            </SelectContent>
   413→          </Select>
   414→          <div className="sm:col-span-1 flex gap-1">
   415→            <Button
   416→              size="sm"
   417→              variant="default"
   418→              className="h-8 w-full"
   419→              onClick={applyFilters}
   420→              disabled={isFetching}
   421→            >
   422→              OK
   423→            </Button>
   424→          </div>
   425→        </div>
   426→        {activeFiltersCount > 0 && (
   427→          <div className="flex flex-wrap gap-1.5 pt-1 items-center">
   428→            {applied.deviceId && (
   429→              <Badge variant="outline" className="text-[10px] gap-1">
   430→                device: {applied.deviceId}
   431→                <button onClick={() => setApplied((a) => ({ ...a, deviceId: undefined }))}>
   432→                  <X className="h-2.5 w-2.5" />
   433→                </button>
   434→              </Badge>
   435→            )}
   436→            {applied.status && (
   437→              <Badge variant="outline" className="text-[10px] gap-1">
   438→                {STATUS_LABEL[applied.status] || applied.status}
   439→                <button onClick={() => setApplied((a) => ({ ...a, status: undefined }))}>
   440→                  <X className="h-2.5 w-2.5" />
   441→                </button>
   442→              </Badge>
   443→            )}
   444→            {applied.routeId && (
   445→              <Badge variant="outline" className="text-[10px] gap-1">
   446→                Маршрут
   447→                <button onClick={() => setApplied((a) => ({ ...a, routeId: undefined }))}>
   448→                  <X className="h-2.5 w-2.5" />
   449→                </button>
   450→              </Badge>
   451→            )}
   452→            <button
   453→              onClick={resetFilters}
   454→              className="text-[10px] text-muted-foreground hover:text-foreground underline ml-1"
   455→            >
   456→              сбросить всё
   457→            </button>
   458→          </div>
   459→        )}
   460→      </div>
   461→
   462→      {/* Список */}
   463→      <div className="flex-1 overflow-y-auto scroll-telem min-h-0">
   464→        {isLoading && !allSessions.length ? (
   465→          <div className="p-3 space-y-2">
   466→            {Array.from({ length: 6 }).map((_, i) => (
   467→              <Skeleton key={i} className="h-16 w-full shimmer" />
   468→            ))}
   469→          </div>
   470→        ) : allSessions.length === 0 ? (
   471→          <div className="p-8 text-center text-sm text-muted-foreground">
   472→            <MapPin className="h-8 w-8 mx-auto mb-2 opacity-30" />
   473→            {isFetching ? "Загрузка…" : "Сессий не найдено"}
   474→            <div className="mt-3">
   475→              <Button variant="outline" size="sm" onClick={() => refetch()}>
   476→                Обновить
   477→              </Button>
   478→            </div>
   479→          </div>
   480→        ) : viewMode === "compact" ? (
   481→          <CompactList
   482→            sessions={sortedSessions}
   483→            selectedId={selectedId}
   484→            onSelect={onSelect}
   485→            bulkMode={bulkMode}
   486→            selectedIds={selectedIds}
   487→            onToggleSelect={toggleSelect}
   488→            onQuickDelete={handleQuickDelete}
   489→            deletingId={deletingId}
   490→          />
   491→        ) : (
   492→          <DetailedList
   493→            groups={grouped || [{ label: "", items: sortedSessions }]}
   494→            selectedId={selectedId}
   495→            onSelect={onSelect}
   496→            bulkMode={bulkMode}
   497→            selectedIds={selectedIds}
   498→            onToggleSelect={toggleSelect}
   499→            onQuickDelete={handleQuickDelete}
   500→            deletingId={deletingId}
   501→          />
   502→        )}
   503→        {hasMore && (
   504→          <div className="p-3 border-t">
   505→            <Button
   506→              variant="outline"
   507→              className="w-full"
   508→              size="sm"
   509→              disabled={isFetching}
   510→              onClick={() => data?.nextCursor && setCursor(data.nextCursor)}
   511→            >
   512→              {isFetching ? (
   513→                <>
   514→                  <Loader2 className="h-3.5 w-3.5 animate-spin" /> Загрузка…
   515→                </>
   516→              ) : (
   517→                "Загрузить ещё"
   518→              )}
   519→            </Button>
   520→          </div>
   521→        )}
   522→      </div>
   523→    </div>
   524→  );
   525→}
   526→
   527→function DetailedList({
   528→  groups,
   529→  selectedId,
   530→  onSelect,
   531→  bulkMode,
   532→  selectedIds,
   533→  onToggleSelect,
   534→  onQuickDelete,
   535→  deletingId,
   536→}: {
   537→  groups: { label: string; items: SessionListItem[] }[];
   538→  selectedId: string | null;
   539→  onSelect: (id: string) => void;
   540→  bulkMode?: boolean;
   541→  selectedIds?: Set<string>;
   542→  onToggleSelect?: (id: string) => void;
   543→  onQuickDelete?: (e: React.MouseEvent, s: SessionListItem) => void;
   544→  deletingId?: string | null;
   545→}) {
   546→  return (
   547→    <>
   548→      {groups.map((group, gi) => (
   549→        <div key={gi}>
   550→          {group.label && (
   551→            <div className="sticky top-0 z-10 bg-muted/60 backdrop-blur-sm px-3 py-1 text-[10px] uppercase tracking-wider text-muted-foreground font-medium border-b">
   552→              {group.label} · {group.items.length}
   553→            </div>
   554→          )}
   555→          <ul className="divide-y">
   556→            {group.items.map((s, idx) => (
