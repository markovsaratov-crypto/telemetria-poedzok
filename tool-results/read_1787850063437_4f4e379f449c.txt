    40→    40→import type { SessionListItem } from "@/lib/api-client";
    41→    41→import { cn } from "@/lib/utils";
    42→    42→
    43→    43→interface SessionsListProps {
    44→    44→  selectedId: string | null;
    45→    45→  onSelect: (id: string) => void;
    46→    46→}
    47→    47→
    48→    48→type SortKey = "date_desc" | "date_asc" | "points_desc" | "size_desc";
    49→    49→type ViewMode = "detailed" | "compact";
    50→    50→
    51→    51→const STATUS_LABEL: Record<string, string> = {
    52→    52→  active: "Активна",
    53→    53→  completed: "Завершена",
    54→    54→  archived: "Архив",
    55→    55→  deleted: "Удалена",
    56→    56→};
    57→    57→
    58→    58→const STATUS_BADGE: Record<string, string> = {
    59→    59→  active: "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/30",
    60→    60→  completed: "bg-teal-500/15 text-teal-700 dark:text-teal-400 border-teal-500/30",
    61→    61→  archived: "bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/30",
    62→    62→  deleted: "bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/30",
    63→    63→};
    64→    64→
    65→    65→const SORT_LABELS: Record<SortKey, string> = {
    66→    66→  date_desc: "Сначала новые",
    67→    67→  date_asc: "Сначала старые",
    68→    68→  points_desc: "Больше точек",
    69→    69→  size_desc: "Больше объём",
    70→    70→};
    71→    71→
    72→    72→export function SessionsList({ selectedId, onSelect }: SessionsListProps) {
    73→    73→  const [deviceId, setDeviceId] = React.useState("");
    74→    74→  const [status, setStatus] = React.useState<string>("");
    75→    75→  const [routeId, setRouteId] = React.useState<string>("");
    76→    76→  const [cursor, setCursor] = React.useState<string | undefined>(undefined);
    77→    77→  const [allSessions, setAllSessions] = React.useState<SessionListItem[]>([]);
    78→    78→  const [hasMore, setHasMore] = React.useState(false);
    79→    79→  const [sort, setSort] = React.useState<SortKey>("date_desc");
    80→    80→  const [viewMode, setViewMode] = React.useState<ViewMode>("detailed");
    81→    81→  const [bulkMode, setBulkMode] = React.useState(false);
    82→    82→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
    83→    83→
    84→    84→  const bulkDeleteMut = useBulkDeleteSessions();
    85→    85→  const deleteMut = useDeleteSession();
    86→    86→  const [deletingId, setDeletingId] = React.useState<string | null>(null);
    87→    87→
    88→    88→  // Применённые фильтры
    89→    89→  const [applied, setApplied] = React.useState<{
    90→    90→    deviceId?: string;
    91→    91→    status?: string;
    92→    92→    routeId?: string;
    93→    93→  }>({});
    94→    94→
    95→    95→  const { data, isLoading, isFetching, refetch } = useSessions({
    96→    96→    limit: 20,
    97→    97→    cursor,
    98→    98→    deviceId: applied.deviceId,
    99→    99→    status: applied.status,
   100→   100→    routeId: applied.routeId,
   101→   101→  });
   102→   102→
   103→   103→  const { data: routesData } = useRoutes();
   104→   104→
   105→   105→  React.useEffect(() => {
   106→   106→    setAllSessions([]);
   107→   107→    setCursor(undefined);
   108→   108→  }, [applied]);
   109→   109→
   110→   110→  React.useEffect(() => {
   111→   111→    if (data) {
   112→   112→      if (cursor) {
   113→   113→        setAllSessions((prev) => [...prev, ...data.sessions]);
   114→   114→      } else {
   115→   115→        setAllSessions(data.sessions);
   116→   116→      }
   117→   117→      setHasMore(!!data.nextCursor);
   118→   118→    }
   119→   119→  }, [data, cursor]);
   120→   120→
   121→   121→  // Сортировка на клиенте (после загрузки)
   122→   122→  const sortedSessions = React.useMemo(() => {
   123→   123→    const arr = [...allSessions];
   124→   124→    arr.sort((a, b) => {
   125→   125→      switch (sort) {
   126→   126→        case "date_asc":
   127→   127→          return new Date(a.startTime).getTime() - new Date(b.startTime).getTime();
   128→   128→        case "points_desc":
   129→   129→          return b.pointCount - a.pointCount;
   130→   130→        case "size_desc":
   131→   131→          return b.payloadBytes - a.payloadBytes;
   132→   132→        default:
   133→   133→          return new Date(b.startTime).getTime() - new Date(a.startTime).getTime();
   134→   134→      }
   135→   135→    });
   136→   136→    return arr;
   137→   137→  }, [allSessions, sort]);
   138→   138→
   139→   139→  // Группировка по дате (только для detailed view)
   140→   140→  const grouped = React.useMemo(() => {
   141→   141→    if (viewMode !== "detailed") return null;
   142→   142→    const groups: { label: string; items: SessionListItem[] }[] = [];
   143→   143→    const today = new Date();
   144→   144→    today.setHours(0, 0, 0, 0);
   145→   145→    const yesterday = new Date(today);
   146→   146→    yesterday.setDate(yesterday.getDate() - 1);
   147→   147→
   148→   148→    for (const s of sortedSessions) {
   149→   149→      const d = new Date(s.startTime);
   150→   150→      d.setHours(0, 0, 0, 0);
   151→   151→      let label: string;
   152→   152→      if (d.getTime() === today.getTime()) label = "Сегодня";
   153→   153→      else if (d.getTime() === yesterday.getTime()) label = "Вчера";
   154→   154→      else
   155→   155→        label = d.toLocaleDateString("ru-RU", {
   156→   156→          day: "numeric",
   157→   157→          month: "long",
   158→   158→          year: d.getFullYear() === today.getFullYear() ? undefined : "numeric",
   159→   159→        });
   160→   160→
   161→   161→      let g = groups.find((g) => g.label === label);
   162→   162→      if (!g) {
   163→   163→        g = { label, items: [] };
   164→   164→        groups.push(g);
   165→   165→      }
   166→   166→      g.items.push(s);
   167→   167→    }
   168→   168→    return groups;
   169→   169→  }, [sortedSessions, viewMode]);
   170→   170→
   171→   171→  function applyFilters() {
   172→   172→    setApplied({
   173→   173→      deviceId: deviceId.trim() || undefined,
   174→   174→      status: status || undefined,
   175→   175→      routeId: routeId || undefined,
   176→   176→    });
   177→   177→  }
   178→   178→
   179→   179→  function resetFilters() {
   180→   180→    setDeviceId("");
   181→   181→    setStatus("");
   182→   182→    setRouteId("");
   183→   183→    setApplied({});
   184→   184→  }
   185→   185→
   186→   186→  function toggleSelect(id: string) {
   187→   187→    setSelectedIds((prev) => {
   188→   188→      const next = new Set(prev);
   189→   189→      if (next.has(id)) next.delete(id);
   190→   190→      else next.add(id);
   191→   191→      return next;
   192→   192→    });
   193→   193→  }
   194→   194→
   195→   195→  function selectAll() {
   196→   196→    setSelectedIds(new Set(allSessions.map((s) => s.id)));
   197→   197→  }
   198→   198→
   199→   199→  function clearSelection() {
   200→   200→    setSelectedIds(new Set());
   201→   201→  }
   202→   202→
   203→   203→  async function handleBulkDelete() {
   204→   204→    if (selectedIds.size === 0) return;
   205→   205→    if (!confirm(`Удалить ${selectedIds.size} сессий? Это soft-delete с grace period 30 дней.`)) {
   206→   206→      return;
   207→   207→    }
   208→   208→    try {
   209→   209→      const result = await bulkDeleteMut.mutateAsync(Array.from(selectedIds));
   210→   210→      toast.success(`Удалено: ${result.deleted}`, {
   211→   211→        description: result.errors.length > 0 ? `${result.errors.length} не найдено` : undefined,
   212→   212→      });
   213→   213→      clearSelection();
   214→   214→      setBulkMode(false);
   215→   215→    } catch (e) {
   216→   216→      toast.error("Ошибка массового удаления", { description: (e as Error).message });
   217→   217→    }
   218→   218→  }
   219→   219→
   220→   220→  async function handleQuickDelete(e: React.MouseEvent, s: SessionListItem) {
   221→   221→    e.stopPropagation();
   222→   222→    e.preventDefault();
   223→   223→    const label = s.deviceName || s.deviceId;
   224→   224→    const date = new Date(s.startTime).toLocaleDateString("ru-RU", { day: "numeric", month: "short", year: "numeric" });
   225→   225→    if (!confirm(`Удалить поездку?\n\n${label} · ${date} · ${fmtNumber(s.pointCount)} точек\n\nЭто soft-delete с grace period 30 дней.`)) {
   226→   226→      return;
   227→   227→    }
   228→   228→    setDeletingId(s.id);
   229→   229→    try {
   230→   230→      await deleteMut.mutateAsync(s.id);
   231→   231→      toast.success("Поездка удалена", {
   232→   232→        description: `${label} · grace period 30 дней, потом purge`,
   233→   233→      });
   234→   234→      // Если удалили текущую выбранную — сбрасываем
   235→   235→      if (selectedId === s.id) {
   236→   236→        onSelect("");
   237→   237→      }
   238→   238→    } catch (err) {
   239→   239→      toast.error("Ошибка удаления", { description: (err as Error).message });
   240→   240→    } finally {
   241→   241→      setDeletingId(null);
   242→   242→    }
   243→   243→  }
   244→   244→
   245→   245→  const activeFiltersCount =
   246→   246→    (applied.deviceId ? 1 : 0) + (applied.status ? 1 : 0) + (applied.routeId ? 1 : 0);
   247→   247→
   248→   248→  return (
   249→   249→    <div className="flex flex-col h-full">
   250→   250→      {/* Header с count и view toggle */}
   251→   251→      <div className="border-b px-3 py-2 flex items-center justify-between bg-muted/30 gap-2">
   252→   252→        <div className="flex items-center gap-2 text-xs">
   253→   253→          <span className="font-medium">
   254→   254→            {allSessions.length > 0 ? `${allSessions.length} сессий` : "Сессии"}
   255→   255→          </span>
   256→   256→          {isFetching && (
   257→   257→            <Loader2 className="h-3 w-3 animate-spin text-muted-foreground" />
   258→   258→          )}
   259→   259→        </div>
   260→   260→        <div className="flex items-center gap-1">
   261→   261→          <Button
   262→   262→            variant="ghost"
   263→   263→            size="sm"
   264→   264→            className="h-7 w-7 p-0"
   265→   265→            onClick={() => refetch()}
   266→   266→            title="Обновить"
   267→   267→          >
   268→   268→            <RefreshCw className="h-3 w-3" />
   269→   269→          </Button>
   270→   270→          <div className="flex items-center rounded-md border bg-background/50 p-0.5">
   271→   271→            <button
   272→   272→              onClick={() => setViewMode("detailed")}
   273→   273→              className={cn(
   274→   274→                "p-1 rounded transition-colors",
   275→   275→                viewMode === "detailed" ? "bg-primary/15 text-primary" : "text-muted-foreground hover:text-foreground"
   276→   276→              )}
   277→   277→              title="Детальный вид"
   278→   278→            >
   279→   279→              <Rows3 className="h-3 w-3" />
   280→   280→            </button>
   281→   281→            <button
   282→   282→              onClick={() => setViewMode("compact")}
   283→   283→              className={cn(
   284→   284→                "p-1 rounded transition-colors",
   285→   285→                viewMode === "compact" ? "bg-primary/15 text-primary" : "text-muted-foreground hover:text-foreground"
   286→   286→              )}
   287→   287→              title="Компактный вид"
   288→   288→            >
   289→   289→              <LayoutList className="h-3 w-3" />
   290→   290→            </button>
   291→   291→          </div>
   292→   292→          <Button
   293→   293→            variant="ghost"
   294→   294→            size="sm"
   295→   295→            className="h-7 gap-1 text-xs"
   296→   296→            onClick={() => {
   297→   297→              setBulkMode((v) => !v);
   298→   298→              if (bulkMode) clearSelection();
   299→   299→            }}
   300→   300→            title={bulkMode ? "Выйти из режима выбора" : "Массовое удаление"}
   301→   301→          >
   302→   302→            {bulkMode ? (
   303→   303→              <>
   304→   304→                <X className="h-3 w-3" /> Отмена
   305→   305→              </>
   306→   306→            ) : (
   307→   307→              <>
   308→   308→                <CheckSquare className="h-3 w-3" /> Выбрать
   309→   309→              </>
   310→   310→            )}
   311→   311→          </Button>
   312→   312→        </div>
   313→   313→      </div>
   314→   314→
   315→   315→      {/* Bulk actions bar */}
   316→   316→      {bulkMode && (
   317→   317→        <motion.div
   318→   318→          initial={{ opacity: 0, height: 0 }}
   319→   319→          animate={{ opacity: 1, height: "auto" }}
   320→   320→          exit={{ opacity: 0, height: 0 }}
   321→   321→          className="border-b bg-amber-500/5 px-3 py-2 flex items-center justify-between gap-2"
   322→   322→        >
   323→   323→          <div className="flex items-center gap-2 text-xs">
   324→   324→            <span className="font-medium text-amber-700 dark:text-amber-400">
   325→   325→              Выбрано: {selectedIds.size}
   326→   326→            </span>
   327→   327→            <button onClick={selectAll} className="text-[10px] underline text-muted-foreground hover:text-foreground">
   328→   328→              все
   329→   329→            </button>
   330→   330→            <button onClick={clearSelection} className="text-[10px] underline text-muted-foreground hover:text-foreground">
   331→   331→              очистить
   332→   332→            </button>
   333→   333→          </div>
   334→   334→          <Button
   335→   335→            size="sm"
   336→   336→            variant="destructive"
   337→   337→            className="h-7 gap-1 text-xs"
   338→   338→            disabled={selectedIds.size === 0 || bulkDeleteMut.isPending}
   339→   339→            onClick={handleBulkDelete}
   340→   340→          >
   341→   341→            {bulkDeleteMut.isPending ? (
   342→   342→              <Loader2 className="h-3 w-3 animate-spin" />
   343→   343→            ) : (
   344→   344→              <Trash2 className="h-3 w-3" />
   345→   345→            )}
   346→   346→            Удалить ({selectedIds.size})
   347→   347→          </Button>
   348→   348→        </motion.div>
   349→   349→      )}
   350→   350→
   351→   351→      {/* Фильтры */}
   352→   352→      <div className="space-y-2 border-b p-3 bg-muted/20">
   353→   353→        <div className="flex items-center justify-between">
   354→   354→          <div className="flex items-center gap-2 text-xs font-medium text-muted-foreground">
   355→   355→            <Filter className="h-3.5 w-3.5" /> Фильтры
   356→   356→            {activeFiltersCount > 0 && (
   357→   357→              <Badge className="text-[9px] h-4 px-1 bg-primary text-primary-foreground">
   358→   358→                {activeFiltersCount}
   359→   359→              </Badge>
   360→   360→            )}
   361→   361→          </div>
   362→   362→          {/* Sort */}
   363→   363→          <Select value={sort} onValueChange={(v) => setSort(v as SortKey)}>
   364→   364→            <SelectTrigger className="h-7 w-[140px] text-[10px] gap-1">
   365→   365→              <ArrowDownUp className="h-2.5 w-2.5" />
   366→   366→              <SelectValue />
   367→   367→            </SelectTrigger>
   368→   368→            <SelectContent>
   369→   369→              {(Object.keys(SORT_LABELS) as SortKey[]).map((k) => (
   370→   370→                <SelectItem key={k} value={k} className="text-xs">
   371→   371→                  {SORT_LABELS[k]}
   372→   372→                </SelectItem>
   373→   373→              ))}
   374→   374→            </SelectContent>
   375→   375→          </Select>
   376→   376→        </div>
   377→   377→        <div className="grid grid-cols-1 sm:grid-cols-12 gap-2">
   378→   378→          <div className="sm:col-span-5 relative">
   379→   379→            <Search className="absolute left-2 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   380→   380→            <Input
   381→   381→              placeholder="Device ID…"
   382→   382→              value={deviceId}
   383→   383→              onChange={(e) => setDeviceId(e.target.value)}
   384→   384→              className="pl-7 h-8 text-xs"
   385→   385→              onKeyDown={(e) => {
   386→   386→                if (e.key === "Enter") applyFilters();
   387→   387→              }}
   388→   388→            />
   389→   389→          </div>
   390→   390→          <Select value={status || "all"} onValueChange={(v) => setStatus(v === "all" ? "" : v)}>
   391→   391→            <SelectTrigger className="sm:col-span-3 h-8 text-xs">
   392→   392→              <SelectValue placeholder="Статус" />
   393→   393→            </SelectTrigger>
   394→   394→            <SelectContent>
   395→   395→              <SelectItem value="all">Все</SelectItem>
   396→   396→              <SelectItem value="active">Активна</SelectItem>
   397→   397→              <SelectItem value="completed">Завершена</SelectItem>
   398→   398→              <SelectItem value="archived">Архив</SelectItem>
   399→   399→            </SelectContent>
   400→   400→          </Select>
   401→   401→          <Select value={routeId || "all"} onValueChange={(v) => setRouteId(v === "all" ? "" : v)}>
   402→   402→            <SelectTrigger className="sm:col-span-3 h-8 text-xs">
   403→   403→              <SelectValue placeholder="Маршрут" />
   404→   404→            </SelectTrigger>
   405→   405→            <SelectContent>
   406→   406→              <SelectItem value="all">Все маршруты</SelectItem>
   407→   407→              {routesData?.routes.map((r) => (
   408→   408→                <SelectItem key={r.id} value={r.id}>
   409→   409→                  {r.name}
   410→   410→                </SelectItem>
   411→   411→              ))}
   412→   412→            </SelectContent>
   413→   413→          </Select>
   414→   414→          <div className="sm:col-span-1 flex gap-1">
   415→   415→            <Button
   416→   416→              size="sm"
   417→   417→              variant="default"
   418→   418→              className="h-8 w-full"
   419→   419→              onClick={applyFilters}
   420→   420→              disabled={isFetching}
   421→   421→            >
   422→   422→              OK
   423→   423→            </Button>
   424→   424→          </div>
   425→   425→        </div>
   426→   426→        {activeFiltersCount > 0 && (
   427→   427→          <div className="flex flex-wrap gap-1.5 pt-1 items-center">
   428→   428→            {applied.deviceId && (
   429→   429→              <Badge variant="outline" className="text-[10px] gap-1">
   430→   430→                device: {applied.deviceId}
   431→   431→                <button onClick={() => setApplied((a) => ({ ...a, deviceId: undefined }))}>
   432→   432→                  <X className="h-2.5 w-2.5" />
   433→   433→                </button>
   434→   434→              </Badge>
   435→   435→            )}
   436→   436→            {applied.status && (
   437→   437→              <Badge variant="outline" className="text-[10px] gap-1">
   438→   438→                {STATUS_LABEL[applied.status] || applied.status}
   439→   439→                <button onClick={() => setApplied((a) => ({ ...a, status: undefined }))}>
   440→   440→                  <X className="h-2.5 w-2.5" />
   441→   441→                </button>
   442→   442→              </Badge>
   443→   443→            )}
   444→   444→            {applied.routeId && (
   445→   445→              <Badge variant="outline" className="text-[10px] gap-1">
   446→   446→                Маршрут
   447→   447→                <button onClick={() => setApplied((a) => ({ ...a, routeId: undefined }))}>
   448→   448→                  <X className="h-2.5 w-2.5" />
   449→   449→                </button>
   450→   450→              </Badge>
   451→   451→            )}
   452→   452→            <button
   453→   453→              onClick={resetFilters}
   454→   454→              className="text-[10px] text-muted-foreground hover:text-foreground underline ml-1"

   520→   520→          </div>
   521→   521→        )}
   522→   522→      </div>
   523→   523→    </div>
